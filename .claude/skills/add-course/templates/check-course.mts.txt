// Builds ONE course folder exactly as pnpm content:build does and prints what a learner will read.
//   cp .claude/skills/add-course/templates/check-course.mts.txt "$SCRATCH/check-course.mts"
//   KS_ROOT="$PWD" pnpm exec tsx "$SCRATCH/check-course.mts" content/courses/<id>
// The folder name is the course id. Its lines come from content/tree. Each move
// shows whose it is, the notes a learner sees, and the chapters that write it out (from <chapter>). Exit 1 on error.
import { existsSync } from 'node:fs';
import { join } from 'node:path';

const root = process.env.KS_ROOT;
const dir = process.argv[2];
if (!root || !dir) throw new Error('Usage: KS_ROOT=<repo> tsx check-course.mts <course-folder>');
const core = await import(join(root, 'packages/core/src/index.ts'));
const lib = await import(join(root, 'scripts/content-lib.ts'));

try {
  const content = lib.loadCourseAt(dir);
  const own = join(dir, 'tree');
  const tree = lib.loadTree(existsSync(own) ? own : undefined);
  for (const variant of content.opening.variants) {
    const line = tree.lines.get(variant.id);
    const flags = [variant.weight ?? variant.whose, variant.alternative ? 'alternative' : '', variant.section ?? '']
      .filter(Boolean)
      .join(', ');
    console.log(`\n${variant.kind} ${variant.id} [${flags}] ${line?.source ?? ''}: ${variant.name}`);
    core.lineSteps(content, variant.id).forEach((step: any, ply: number) => {
      const n = `${Math.floor(ply / 2) + 1}${ply % 2 ? '...' : '.'}`;
      const notes = core.explainMove(step.fromEpd, step.edge.uci, { edge: step.edge });
      const writers = tree.edges.get(`${step.fromEpd} ${step.edge.uci}`)?.writtenIn ?? [];
      const from = writers.length === 1 && writers[0] === variant.id ? '' : `  (from ${writers.join(', ')})`;
      const missing = notes.length ? '' : step.ours ? '  [no note]' : '';
      console.log(`  ${n} ${step.edge.san} (${step.ours ? 'you' : 'them'})${from}${missing}`);
      for (const note of notes) console.log(`      ${note.source}: ${note.text}`);
      const text = step.edge.annotation?.text ?? '';
      if (text.length > 220) console.log(`      WARN ${text.length} characters; aim for under about 200`);
    });
  }
  const cards = core.introductionOrder(content, 0);
  console.log(
    `\nOK ${content.opening.id}: ${Object.keys(content.nodes).length} positions, ${cards.length} cards incl. ${content.quizzes.length} quizzes.`,
  );
  console.log(
    'Not checked here: openings.yaml, motifs.yaml, curriculum.yaml and other courses. Run pnpm content:build for those.',
  );
} catch (error) {
  console.error(`FAILED: ${error instanceof Error ? error.message : error}`);
  process.exit(1);
}
